Scope 2 emissions from electricity use

시간별 전력 사용량에 배출계수를 곱해 Scope 2 배출량을 계산하는 예시.

import pandas as pd

def scope2_emissions(usage_kwh: pd.Series, grid_ef: float) -> float:
    """Calculate location-based Scope 2 emissions.

    usage_kwh : hourly electricity consumption (kWh)
    grid_ef   : grid emission factor (kgCO2e/kWh)
    """
    return usage_kwh.sum() * grid_ef

df = pd.read_csv("datacentre_usage.csv", parse_dates=["timestamp"])
total = scope2_emissions(df["kwh"], grid_ef=0.412)
print(f"Annual Scope 2: {total / 1000:.1f} tCO2e")
